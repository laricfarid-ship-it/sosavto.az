const ads = allAds.filter(ad => { // Əgər VIN yazıbsa, digər kateqoriya/marka məhdudiyyətlərini keçib birbaşa VIN ilə axtarış edirik if (vinVal) { const adVin = (ad.vin || "").toLowerCase(); return adVin.includes(vinVal);
