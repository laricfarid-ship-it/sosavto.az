async function loadAds() { const grid = document.getElementById('adsGrid'); const counter = document.getElementById('adCounter'); const allAds = JSON.parse(localStorage.getItem('sosavto_ads')) || []; const vinVal = document.getElementById('filterVin').value.trim(); let ads = allAds.filter(ad => { if (!vinVal) return true; return (ad.vin || "").toLowerCase().includes(vinVal.toLowerCase()); }); let externalCar = null; if (ads.length === 0 && vinVal.length >= 5) { try { const response = await fetch(`https://vpic.nhtsa.dot.gov/api/vehicles/decodevinvalues/${vinVal}?format=json`); const data = await response.json(); if (data && data.Results && data.Results.length > 0) { const res = data.Results[0]; if (res.Make && res.Make !== "") { externalCar = { title: `${res.Make} ${res.Model || ''} (${res.ModelYear || 'Naməlum il'})`, autoBrand: res.Make, autoModel: res.Model || 'Model göstərilməyib', autoYear: res.ModelYear || '', vin: vinVal, category: 'auto', address: `${res.PlantCountry || 'Baza məlumatı'}, ${res.VehicleType || 'Avtomobil'}`, autoPrice: 'Qlobal Baza', description: `Zavod məlumatları: Mühərrik: ${res.DisplacementL || '-'}L, Yanacaq: ${res.FuelTypePrimary || '-'}` }; } } } catch (e) { console.error("Baza xətası:", e); } } let displayList = [...ads]; if (externalCar) { displayList.push(externalCar); } counter.textContent = `Cəmi ${displayList.length} nəticə tapıldı`; grid.innerHTML = ''; if (displayList.length === 0) { grid.innerHTML = `

Axtarışa uyğun nəticə tapılmadı.

`; return; } displayList.reverse().forEach(ad => { let badgeText = ad.autoPrice === 'Qlobal Baza' ? 'Rəsmi Baza' : 'Avtomobil'; let badgeClass = ad.autoPrice === 'Qlobal Baza' ? 'bg-emerald-500/20 text-emerald-400 border-emerald-500/30' : 'bg-red-500/20 text-red-400 border-red-500/30'; let priceText = ad.autoPrice === 'Qlobal Baza' ? 'Rəsmi Yoxlama' : `${ad.autoPrice} AZN`; let subInfo = `${ad.autoBrand || ''} ${ad.autoModel || ''} (${ad.autoYear || ''})`; const imageHtml = ad.image ? `` : `
`; const card = document.createElement('div'); card.className = 'bg-gray-900/90 border border-gray-800 rounded-2xl overflow-hidden shadow-lg hover:border-amber-500/50 transition flex flex-col justify-between cursor-pointer group'; card.onclick = () => openModal(ad); card.innerHTML = `
${imageHtml} ${badgeText}
${priceText}

${ad.title}

VIN: ${ad.vin || vinVal}

${subInfo}

Ətraflı Bax
`; grid.appendChild(card); }); }
